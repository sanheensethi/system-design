4.1 LRU Cache

A cache with fixed capacity that evicts the least-recently-used entry. The archetypal Google interview question.

🎬 Interactive: Watch put/get operations with live state Open full viz ↗

v1 — List scan O(n)

class LRU:
    def __init__(self, cap):
        self.cap = cap
        self.data = []  # list of (key, value); most recent at end

    def get(self, k):
        for i, (kk, v) in enumerate(self.data):
            if kk == k:
                self.data.pop(i)
                self.data.append((k, v))
                return v
        return -1

    def put(self, k, v):
        for i, (kk, _) in enumerate(self.data):
            if kk == k:
                self.data.pop(i); break
        self.data.append((k, v))
        if len(self.data) > self.cap:
            self.data.pop(0)

Pain: O(n) scans on every operation. A 10K-entry cache at 100K ops/sec is dead.

v2 — HashMap + sorted list by timestamp

import time
class LRU:
    def __init__(self, cap):
        self.cap = cap
        self.map = {}  # k -> (v, ts)

    def get(self, k):
        if k not in self.map: return -1
        v, _ = self.map[k]
        self.map[k] = (v, time.monotonic_ns())
        return v

    def put(self, k, v):
        self.map[k] = (v, time.monotonic_ns())
        if len(self.map) > self.cap:
            # evict oldest -> still O(n) to find min
            oldest = min(self.map, key=lambda x: self.map[x][1])
            del self.map[oldest]

Pain: Eviction is still O(n). We need true O(1) for all operations.

v3 — HashMap + Doubly Linked List (the canonical answer)

class Node:
    __slots__ = ("k", "v", "prev", "nxt")
    def __init__(self, k, v):
        self.k, self.v = k, v
        self.prev = self.nxt = None

class LRU:
    def __init__(self, cap):
        self.cap = cap
        self.map = {}
        # sentinels: head <-> ... <-> tail
        # head.next = MRU, tail.prev = LRU
        self.head = Node(0, 0)
        self.tail = Node(0, 0)
        self.head.nxt = self.tail
        self.tail.prev = self.head

    def _remove(self, n):
        n.prev.nxt = n.nxt
        n.nxt.prev = n.prev

    def _add_front(self, n):
        n.nxt = self.head.nxt
        n.prev = self.head
        self.head.nxt.prev = n
        self.head.nxt = n

    def get(self, k):
        if k not in self.map: return -1
        n = self.map[k]
        self._remove(n); self._add_front(n)
        return n.v

    def put(self, k, v):
        if k in self.map:
            n = self.map[k]
            n.v = v
            self._remove(n); self._add_front(n)
            return
        n = Node(k, v)
        self.map[k] = n
        self._add_front(n)
        if len(self.map) > self.cap:
            lru = self.tail.prev
            self._remove(lru)
            del self.map[lru.k]
head <-> [MRU] <-> [...] <-> [LRU] <-> tail get(k): unlink node, re-insert after head evict: remove tail.prev

Pain: Two threads call get simultaneously, both mutate the doubly linked list, pointers go wild.

v4 — Thread-safe with RLock

import threading
class LRU:
    def __init__(self, cap):
        self.cap = cap
        self.map = {}
        self.head = Node(0, 0); self.tail = Node(0, 0)
        self.head.nxt = self.tail; self.tail.prev = self.head
        self.lock = threading.RLock()

    def get(self, k):
        with self.lock:
            if k not in self.map: return -1
            n = self.map[k]
            self._remove(n); self._add_front(n)
            return n.v

    def put(self, k, v):
        with self.lock:
            # ... same as v3
            ...

Pain: One big lock serializes all reads. Under heavy concurrency the cache becomes a bottleneck.

v5 — Striped locks (sharded cache)

class StripedLRU:
    def __init__(self, cap, stripes=16):
        per = max(1, cap // stripes)
        self.shards = [LRU(per) for _ in range(stripes)]

    def _shard(self, k):
        return self.shards[hash(k) % len(self.shards)]

    def get(self, k): return self._shard(k).get(k)
    def put(self, k, v): self._shard(k).put(k, v)

Contention divides by stripe count. Caveat: true global LRU ordering is lost — each shard has its own LRU, which is acceptable for most caching workloads.

4.2 Rate Limiter

v1 — Count per minute bucket

import time
class RateLimiter:
    def __init__(self, limit):
        self.limit = limit
        self.window = int(time.time()) // 60
        self.count = 0

    def allow(self):
        now = int(time.time()) // 60
        if now != self.window:
            self.window = now; self.count = 0
        if self.count < self.limit:
            self.count += 1; return True
        return False

Pain: At 10:59:59 a burst of N requests, at 11:00:00 another burst of N. That's 2N in one second. Classic fixed-window edge.

v2 — Sliding window log

from collections import deque
class SlidingWindow:
    def __init__(self, limit, window_sec):
        self.limit, self.win = limit, window_sec
        self.q = deque()

    def allow(self):
        now = time.time()
        while self.q and now - self.q[0] > self.win:
            self.q.popleft()
        if len(self.q) < self.limit:
            self.q.append(now); return True
        return False

Pain: Memory grows with request rate — we keep a timestamp per request. A client hitting 10K rps costs 10K floats of RAM per key.

v3 — Token bucket

class TokenBucket:
    def __init__(self, rate, capacity):
        self.rate = rate        # tokens per second
        self.cap = capacity
        self.tokens = capacity
        self.ts = time.monotonic()

    def allow(self, cost=1):
        now = time.monotonic()
        elapsed = now - self.ts
        self.tokens = min(self.cap, self.tokens + elapsed * self.rate)
        self.ts = now
        if self.tokens >= cost:
            self.tokens -= cost; return True
        return False

Fixed memory per key, allows bursts up to capacity, smooths long-term to rate. This is what AWS, Stripe, and most APIs actually use.

Pain: Single-process only. Behind a load balancer with 10 app servers, each enforces its own limit. Effective rate = 10× intended.

v4 — Distributed via Redis + Lua (atomic)

LUA = """
local key = KEYS[1]
local rate = tonumber(ARGV[1])
local cap  = tonumber(ARGV[2])
local now  = tonumber(ARGV[3])
local cost = tonumber(ARGV[4])

local data = redis.call('HMGET', key, 'tokens', 'ts')
local tokens = tonumber(data[1]) or cap
local ts = tonumber(data[2]) or now

local elapsed = math.max(0, now - ts)
tokens = math.min(cap, tokens + elapsed * rate)

local allowed = 0
if tokens >= cost then
    tokens = tokens - cost
    allowed = 1
end

redis.call('HMSET', key, 'tokens', tokens, 'ts', now)
redis.call('EXPIRE', key, math.ceil(cap / rate) * 2)
return allowed
"""

class RedisTokenBucket:
    def __init__(self, redis, rate, cap):
        self.r, self.rate, self.cap = redis, rate, cap
        self.sha = redis.script_load(LUA)

    def allow(self, key, cost=1):
        return bool(self.r.evalsha(
            self.sha, 1, key, self.rate, self.cap, time.time(), cost))

Lua runs atomically inside Redis — no two app servers can interleave a read-modify-write on the same key.

Pain: Now the team wants to A/B test sliding window vs token bucket. Different endpoints need different algorithms.

v5 — Strategy pattern

class LimiterStrategy(ABC):
    @abstractmethod
    def allow(self, key) -> bool: ...

class TokenBucketStrategy(LimiterStrategy): ...
class SlidingWindowStrategy(LimiterStrategy): ...
class LeakyBucketStrategy(LimiterStrategy): ...

class Gateway:
    def __init__(self):
        self.routes = {
            "/login": SlidingWindowStrategy(5, 60),
            "/api/":  TokenBucketStrategy(rate=100, cap=200),
        }
    def allow(self, path, user):
        for prefix, strat in self.routes.items():
            if path.startswith(prefix):
                return strat.allow(user)
        return True

4.3 Elevator System

v1 — Naive FIFO queue

from collections import deque
class Elevator:
    def __init__(self):
        self.floor = 0
        self.q = deque()

    def request(self, f):
        self.q.append(f)

    def step(self):
        if not self.q: return
        target = self.q[0]
        self.floor += 1 if target > self.floor else -1
        if self.floor == target: self.q.popleft()

Pain: If requests arrive as [5, 1, 6], elevator goes 0→5→1→6. Wasteful. Real elevators use SCAN (sweep up, then down).

v2 — Directional scheduling (SCAN)

class ScanElevator:
    def __init__(self):
        self.floor = 0
        self.direction = 0  # -1 down, 0 idle, +1 up
        self.up_set = set()
        self.down_set = set()

    def request(self, f):
        if f > self.floor: self.up_set.add(f)
        elif f < self.floor: self.down_set.add(f)

    def step(self):
        if self.direction == 0:
            self.direction = 1 if self.up_set else (-1 if self.down_set else 0)
        if self.direction == 1:
            if not self.up_set:
                self.direction = -1 if self.down_set else 0; return
            self.floor += 1
            self.up_set.discard(self.floor)
        elif self.direction == -1:
            if not self.down_set:
                self.direction = 1 if self.up_set else 0; return
            self.floor -= 1
            self.down_set.discard(self.floor)

Pain: Works for one car. A real building has 4–8 elevators. Need dispatch.

v3 — Multiple elevators with load balancing

class Dispatcher:
    def __init__(self, cars):
        self.cars = cars

    def assign(self, floor, direction):
        # score each elevator: lower = better
        def score(e):
            if e.direction == 0:
                return abs(e.floor - floor)
            # same direction and along the way = ideal
            if e.direction == direction:
                along = (direction == 1 and floor >= e.floor) or \
                        (direction == -1 and floor <= e.floor)
                return abs(e.floor - floor) if along else 1000 + abs(e.floor - floor)
            # opposite direction = must finish sweep first
            return 500 + abs(e.floor - floor)
        best = min(self.cars, key=score)
        best.request(floor)
        return best

Pain: Each elevator still has a blob of state flags (direction, maintenance, door_open, emergency). State transitions scattered across many if checks.

v4 — State pattern per elevator

class ElevatorState(ABC):
    @abstractmethod
    def step(self, e): ...

class Idle(ElevatorState):
    def step(self, e):
        if e.up_set: e.set_state(GoingUp())
        elif e.down_set: e.set_state(GoingDown())

class GoingUp(ElevatorState):
    def step(self, e):
        if not e.up_set:
            e.set_state(GoingDown() if e.down_set else Idle()); return
        e.floor += 1
        if e.floor in e.up_set:
            e.up_set.remove(e.floor)
            e.open_doors()

class GoingDown(ElevatorState):
    def step(self, e):
        if not e.down_set:
            e.set_state(GoingUp() if e.up_set else Idle()); return
        e.floor -= 1
        if e.floor in e.down_set:
            e.down_set.remove(e.floor)
            e.open_doors()

class Maintenance(ElevatorState):
    def step(self, e): pass  # frozen

v5 — Observer for floor requests / display panels

class FloorPanel:
    def __init__(self, floor, dispatcher):
        self.floor, self.dispatcher = floor, dispatcher
        self.observers = []

    def press_up(self):
        car = self.dispatcher.assign(self.floor, +1)
        for o in self.observers:
            o.on_call_accepted(self.floor, car)

class FloorDisplay:
    def on_car_move(self, car):
        print(f"Car {car.id} at floor {car.floor}")

class EmergencyLogger:
    def on_emergency(self, car): pager.alert(car.id)

4.4 Splitwise

v1 — Direct IOU list

class Splitwise:
    def __init__(self):
        self.ious = []  # (debtor, creditor, amount)

    def add(self, payer, participants, amount):
        each = amount / len(participants)
        for p in participants:
            if p != payer:
                self.ious.append((p, payer, each))

    def what_owes(self, user):
        owed = sum(a for d, c, a in self.ious if d == user)
        owes = sum(a for d, c, a in self.ious if c == user)
        return owed - owes

Pain: 1000 people, 50K transactions — a single "who owes whom" query is O(n). Also, A→B→C→A loops don't cancel.

v2 — Balance matrix

from collections import defaultdict
class Splitwise:
    def __init__(self):
        # balance[a][b] = amount a owes b
        self.balance = defaultdict(lambda: defaultdict(float))

    def add(self, payer, participants, amount):
        each = amount / len(participants)
        for p in participants:
            if p == payer: continue
            # net out if already owes in reverse
            self.balance[p][payer] += each
            if self.balance[payer][p] > 0:
                d = min(self.balance[p][payer], self.balance[payer][p])
                self.balance[p][payer] -= d
                self.balance[payer][p] -= d

Pain: Cycles A→B→C→A still waste settlements. 3 transfers when 0 are needed.

v3 — Graph with debt simplification

import heapq
class DebtSimplifier:
    def simplify(self, transactions):
        # 1. compute net balance per person
        net = defaultdict(float)
        for payer, parts, amt in transactions:
            each = amt / len(parts)
            for p in parts:
                net[p] -= each
            net[payer] += amt

        # 2. greedy: pair largest creditor with largest debtor
        creditors = [(-v, u) for u, v in net.items() if v > 0.01]
        debtors   = [(v, u) for u, v in net.items() if v < -0.01]
        heapq.heapify(creditors); heapq.heapify(debtors)

        settlements = []
        while creditors and debtors:
            cv, cu = heapq.heappop(creditors)
            dv, du = heapq.heappop(debtors)
            pay = min(-cv, -dv)
            settlements.append((du, cu, pay))
            if -cv - pay > 0.01:
                heapq.heappush(creditors, (cv + pay, cu))
            if -dv - pay > 0.01:
                heapq.heappush(debtors, (dv + pay, du))
        return settlements

A→B→C→A with equal amounts collapses to zero transfers. Minimum cash-settlement problem (NP-hard in general, greedy is usually fine in practice).

Pain: Only equal splits supported. Real groups need "exact amounts", "by percentage", "by shares".

v4 — Strategy for split types

class SplitStrategy(ABC):
    @abstractmethod
    def split(self, total, participants, meta) -> dict: ...

class EqualSplit(SplitStrategy):
    def split(self, total, participants, meta):
        each = total / len(participants)
        return {p: each for p in participants}

class ExactSplit(SplitStrategy):
    def split(self, total, participants, meta):
        # meta[p] = exact amount
        assert abs(sum(meta.values()) - total) < 0.01
        return dict(meta)

class PercentSplit(SplitStrategy):
    def split(self, total, participants, meta):
        # meta[p] = percent
        assert abs(sum(meta.values()) - 100) < 0.01
        return {p: total * meta[p] / 100 for p in participants}

class ShareSplit(SplitStrategy):
    def split(self, total, participants, meta):
        # meta[p] = share units (e.g., 2 roommates + 1 kid = 2,2,1)
        units = sum(meta.values())
        return {p: total * meta[p] / units for p in participants}

v6 — Group expenses

class Group:
    def __init__(self, name, members):
        self.name, self.members = name, set(members)
        self.expenses = []
        self.simplifier = DebtSimplifier()
        self.observers = []

    def add_expense(self, payer, amount, strategy, participants=None, meta=None):
        parts = participants or self.members
        shares = strategy.split(amount, parts, meta or {})
        e = Expense(payer, amount, shares)
        self.expenses.append(e)
        for o in self.observers: o.on_expense(e)

        return self.simplifier.simplify(
            [(e.payer, list(e.shares), e.amount) for e in self.expenses])

4.5 Library Management

v1 — Dict of books

class Library:
    def __init__(self):
        self.books = {}  # isbn -> count

    def add(self, isbn): self.books[isbn] = self.books.get(isbn, 0) + 1
    def borrow(self, isbn):
        if self.books.get(isbn, 0) > 0:
            self.books[isbn] -= 1; return True
        return False

Pain: No idea who borrowed what, no due dates, no fines.

v2 — Book / Member / Transaction classes

from datetime import datetime, timedelta

class Book:
    def __init__(self, isbn, title, author):

    def borrow(self, member, isbn):

class BookBase(ABC):

class Magazine(BookBase):

v5 — Strategy for fine calculation

class FineStrategy(ABC):
    @abstractmethod
    def calculate(self, transaction, returned_at) -> float: ...

class FlatDailyFine(FineStrategy):
    def __init__(self, per_day): self.per_day = per_day
    def calculate(self, t, ret):
        days = max(0, (ret - t.due).days)
        return days * self.per_day

class TieredFine(FineStrategy):
    """1–7 days: $1/day, 8–14: $2/day, 15+: $5/day."""
    def calculate(self, t, ret):
        days = max(0, (ret - t.due).days)
        fine, rem = 0, days
        for limit, rate in [(7, 1), (7, 2), (float('inf'), 5)]:
            used = min(rem, limit)
            fine += used * rate; rem -= used
            if rem <= 0: break
        return fine

class MembershipTierFine(FineStrategy):
    """Premium members: free for 3 days, then half rate."""
    def __init__(self, base: FineStrategy): self.base = base
    def calculate(self, t, ret):
        raw = self.base.calculate(t, ret)
        if t.member.tier == "premium":
            grace_days = 3
            effective_due = t.due + timedelta(days=grace_days)
            if ret <= effective_due: return 0
            return raw / 2
        return raw
